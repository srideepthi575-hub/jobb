const CACHE_KEY = "JOB_RADAR_CACHE_DATA";
const TIMESTAMP_KEY = "JOB_RADAR_CACHE_TIME";
const TWENTY_FOUR_HOURS = 24 * 60 * 60 * 1000;

document.addEventListener("DOMContentLoaded", () => {
  loadJobsWith24HourLogic();

  document.getElementById("searchBtn").addEventListener("click", () => {
    const query = document.getElementById("searchInput").value.trim();
    fetchAllRealSources(query);
  });

  document.getElementById("forceRefreshBtn").addEventListener("click", () => {
    const query = document.getElementById("searchInput").value.trim();
    fetchAllRealSources(query, true);
  });
});

async function loadJobsWith24HourLogic() {
  const cachedData = localStorage.getItem(CACHE_KEY);
  const cachedTime = localStorage.getItem(TIMESTAMP_KEY);
  const now = new Date().getTime();

  if (cachedData && cachedTime && (now - Number(cachedTime) < TWENTY_FOUR_HOURS)) {
    const parsedJobs = JSON.parse(cachedData);
    const nextUpdateInHours = Math.round((TWENTY_FOUR_HOURS - (now - Number(cachedTime))) / (1000 * 60 * 60));
    
    renderJobsUI(parsedJobs);
    document.getElementById("jobCountText").textContent = `Showing ${parsedJobs.length} cached active jobs.`;
    document.getElementById("lastUpdatedText").textContent = `Last auto-updated: ${new Date(Number(cachedTime)).toLocaleTimeString()} (Next refresh in ~${nextUpdateInHours} hrs)`;
  } else {
    await fetchAllRealSources();
  }
}

async function fetchAllRealSources(query = "", force = false) {
  const countText = document.getElementById("jobCountText");
  const grid = document.getElementById("jobGrid");

  countText.textContent = "Connecting to live job feeds...";
  grid.innerHTML = `<div class="empty-state"><p>Scanning active sources (Himalayas & Remotive)...</p></div>`;

  try {
    // Source 1: Himalayas API
    const himalayasUrl = query 
      ? `https://himalayas.app/jobs/api/search?q=${encodeURIComponent(query)}`
      : `https://himalayas.app/jobs/api?limit=25`;

    // Source 2: Remotive API
    const remotiveUrl = query
      ? `https://remotive.com/api/remote-jobs?search=${encodeURIComponent(query)}`
      : `https://remotive.com/api/remote-jobs?limit=25`;

    const [himRes, remRes] = await Promise.allSettled([
      fetch(himalayasUrl).then(r => r.json()),
      fetch(remotiveUrl).then(r => r.json())
    ]);

    let combinedJobs = [];

    // Parse Himalayas Data
    if (himRes.status === "fulfilled" && himRes.value.jobs) {
      const himJobs = himRes.value.jobs.map(j => ({
        title: j.title || "Software Engineer",
        company: j.companyName || "Verified Employer",
        location: j.locationRestrictions?.map(l => l.name).join(", ") || "Remote",
        workMode: "Remote",
        jobType: j.employmentType || "Full Time",
        salary: j.minSalary ? `$${j.minSalary.toLocaleString()} - $${j.maxSalary.toLocaleString()}` : "Not specified",
        skills: [...(j.categories || []), ...(j.parentCategories || [])],
        source: "Himalayas",
        sourceUrl: j.applicationLink || "https://himalayas.app"
      }));
      combinedJobs.push(...himJobs);
    }

    // Parse Remotive Data
    if (remRes.status === "fulfilled" && remRes.value.jobs) {
      const remJobs = remRes.value.jobs.map(j => ({
        title: j.title || "Developer Role",
        company: j.company_name || "Tech Company",
        location: j.candidate_required_location || "Worldwide",
        workMode: "Remote",
        jobType: j.job_type || "Full Time",
        salary: j.salary || "Not specified",
        skills: j.category ? [j.category] : ["Software"],
        source: "Remotive",
        sourceUrl: j.url || "https://remotive.com"
      }));
      combinedJobs.push(...remJobs);
    }

    // Deduplication Engine (Title + Company)
    const seen = new Set();
    const uniqueJobs = combinedJobs.filter(j => {
      const key = `${j.title.toLowerCase()}-${j.company.toLowerCase()}`;
      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    });

    if (uniqueJobs.length === 0) {
      countText.textContent = "0 Active Jobs Found";
      grid.innerHTML = `
        <div class="empty-state">
          <h3>No Active Jobs Found</h3>
          <p>No verified live openings match "${escapeHTML(query)}". Try another keyword.</p>
        </div>
      `;
      return;
    }

    if (!query || force) {
      localStorage.setItem(CACHE_KEY, JSON.stringify(uniqueJobs));
      localStorage.setItem(TIMESTAMP_KEY, new Date().getTime().toString());
    }

    renderJobsUI(uniqueJobs);
    countText.textContent = `Displaying ${uniqueJobs.length} Verified Active Openings`;
    document.getElementById("lastUpdatedText").textContent = `Fetched Live: ${new Date().toLocaleTimeString()}`;

  } catch (error) {
    countText.textContent = "Failed to load live feed.";
    grid.innerHTML = `<div class="empty-state"><p>Error connecting to live source endpoints.</p></div>`;
  }
}

function renderJobsUI(jobs) {
  const grid = document.getElementById("jobGrid");
  
  grid.innerHTML = jobs.map(job => `
    <div class="job-card">
      <div class="job-info">
        <h2 class="job-title">${escapeHTML(job.title)}</h2>
        <p class="job-company">${escapeHTML(job.company)} · ${escapeHTML(job.location)}</p>
        <div class="tag-list">
          <span class="tag">${escapeHTML(job.workMode)}</span>
          <span class="tag">${escapeHTML(job.jobType)}</span>
          <span class="tag" style="background:#334155; color:#fff;">Source: ${escapeHTML(job.source)}</span>
          ${job.skills.slice(0, 2).map(skill => `<span class="tag" style="color: #94a3b8;">${escapeHTML(skill)}</span>`).join('')}
        </div>
      </div>
      <div class="job-action">
        <div class="salary-text">${escapeHTML(job.salary)}</div>
        <a href="${job.sourceUrl}" target="_blank" rel="noopener noreferrer" class="apply-btn">Apply Direct</a>
      </div>
    </div>
  `).join('');
}

function escapeHTML(str) {
  return String(str || '').replace(/[&<>"']/g, match => ({
    '&': '&amp;',
    '<': '&lt;',
    '>': '&gt;',
    '"': '&quot;',
    "'": '&#39;'
  }[match]));
}
